### T0: Add the TcpConnection transport and multi-client serving to agentj-ipc [P0] [Sequential]

> **Design alignment**: `docs/superpowers/specs/2026-08-15-agentj-daemon-kernel-design.md` 决策 #1/#3/#4/#5/#9（统一传输抽象 + 按需常驻 + 广播订阅 + 审批定向）——本任务将原「TcpIpcServer 专属骨架」升级为「v0.1 Connection trait 的 TcpConnection 实现 + ConnectionRegistry 多连接路径」。协议加性扩展（`session/subscribe` / `session/unsubscribe` / `$/permissionRequest` origin_client / `-32805 SessionBusy` / `-32806 NotAuthorized`）wire 类型已在 agentj-protocol 落地（守护内核计划 Part A）；本任务在 serve 循环中消费。
> - `TcpConnection` 实现 v0.1 的 `Connection` trait（TCP NDJSON 帧）
> - 统一 serve 循环服务 N 个并发连接（ONE shared kernel——模块生命周期集中）
> - `ConnectionRegistry` 多连接路径：注册/注销、`session_owner` 定向路由、`pending_runs` 互斥
> - `session/subscribe`：`$/progress` 广播给该 session 所有订阅者；`session/unsubscribe` 显式退订；连接断开隐式退订
> - `$/permissionRequest` 按 `session_owner` 定向到发起方；`permission/respond` 校验归属（非 owner → `-32806 NotAuthorized`）
> - 同 session 并发 `session/run` → `-32805 SessionBusy`
> - **权限边缘语义**（设计 错误处理，必须携带）：Ask 审批超时（默认 60s）自动 Deny；owner 断开 → 挂起审批全部 Deny（`PermissionError::RequestCancelled`），run 返回拒绝；subscribe 断开 → 隐式退订无残留；最后连接断开 → 内核优雅关闭链（信号仍可强制）
> - **鉴权/限流定稿**（原 design-pending FINALIZED）：可选 `--token`（客户端 `initialize` 携带，不匹配 → `-32806 NotAuthorized`）；`--max-connections`（默认 16）超限新连接直接拒绝

    # agentj-ipc/src/tcp.rs (new file)
    use std::sync::Arc;
    use std::time::Duration;
    use tokio::io::{AsyncBufReadExt, AsyncWriteExt, BufReader};
    use tokio::net::{TcpListener, TcpStream};
    use agentj_core::agent_loop::AgentLoop;
    use agentj_core::kernel::Kernel;
    use crate::{IpcError, IpcServer};

    /// Multi-connection JSON-RPC over TCP — the Web Dashboard's protocol entry (design 概述:
    /// Web Dashboard 作为 JSON-RPC 协议消费者). ONE shared IpcServer (shared Kernel + AgentLoop)
    /// serves ALL connections, so module lifecycle stays centralized; each connection is a
    /// `TcpConnection` (implements the v0.1 `Connection` trait) running the unified serve loop.
    pub struct TcpIpcServer {
        listener: TcpListener,
        shared: Arc<IpcServer>,
        max_connections: usize,
        token: Option<String>,
    }

    impl TcpIpcServer {
        pub async fn bind(
            addr: &str, agent_loop: AgentLoop, kernel: Kernel,
            token: Option<String>, max_connections: usize,
        ) -> Result<Self, IpcError> {
            let listener = TcpListener::bind(addr).await
                .map_err(|e| IpcError::InvalidRequest(e.to_string()))?;
            // IpcServer::new starts the kernel's event loop; every connection shares it
            let shared = Arc::new(IpcServer::new(agent_loop, kernel));
            Ok(Self { listener, shared, max_connections, token })
        }

        pub fn local_addr(&self) -> Result<std::net::SocketAddr, IpcError> {
            self.listener.local_addr().map_err(|e| IpcError::InvalidRequest(e.to_string()))
        }

        pub async fn run(&mut self) -> Result<(), IpcError> {
            let mut active = 0usize;
            // Graceful shutdown: when the LAST connection disconnects (or any client sends
            // `shutdown` flipping the shared flag), the accept loop exits and the kernel's
            // graceful shutdown chain runs (design 错误处理: 最后连接断开 → 优雅关闭).
            loop {
                tokio::select! {
                    res = self.listener.accept() => {
                        let (stream, _) = res.map_err(|e| IpcError::InvalidRequest(e.to_string()))?;
                        if active >= self.max_connections {
                            tracing::warn!("max connections reached ({})", self.max_connections);
                            drop(stream); // refuse: close the socket (design 错误处理)
                            continue;
                        }
                        let shared = Arc::clone(&self.shared);
                        let token = self.token.clone();
                        active += 1;
                        tokio::spawn(async move {
                            let conn = TcpConnection::new(stream);
                            if let Err(e) = serve_connection(conn, &shared, token.as_deref()).await {
                                tracing::error!(error = %e, "tcp connection error");
                            }
                        });
                    }
                    _ = tokio::time::sleep(Duration::from_millis(50)) => {
                        if self.shared.shutdown_requested() { break; }
                    }
                }
            }
            Ok(())
        }
    }

    /// TCP transport — implements the v0.1 `Connection` trait (design 决策 #1).
    pub struct TcpConnection {
        id: String,
        reader: BufReader<tokio::net::tcp::OwnedReadHalf>,
        writer: Arc<tokio::sync::Mutex<tokio::net::tcp::OwnedWriteHalf>>,
        peer: String,
    }

    impl TcpConnection {
        pub fn new(stream: TcpStream) -> Self {
            let peer = stream.peer_addr().map(|a| a.to_string()).unwrap_or_else(|_| "unknown".into());
            let (r, w) = stream.into_split();
            Self {
                id: format!("tcp-{peer}"),
                reader: BufReader::new(r),
                writer: Arc::new(tokio::sync::Mutex::new(w)),
                peer,
            }
        }
    }

    impl Connection for TcpConnection {
        fn id(&self) -> String { self.id.clone() }
        fn kind(&self) -> TransportKind { TransportKind::Tcp }
        fn send(&self, msg: &str) -> Result<(), IpcError> {
            let writer = self.writer.clone();
            tokio::runtime::Handle::current().block_on(async move {
                let mut w = writer.lock().await;
                w.write_all(msg.as_bytes()).await?;
                w.write_all(b"\n").await?;
                w.flush().await?;
                Ok(())
            }).map_err(|e: std::io::Error| IpcError::InvalidRequest(e.to_string()))
        }
        async fn recv(&mut self) -> Option<Result<String, IpcError>> {
            let mut line = String::new();
            match self.reader.read_line(&mut line).await {
                Ok(0) => None, // client closed
                Ok(_) => Some(Ok(line)),
                Err(e) => Some(Err(IpcError::InvalidRequest(e.to_string()))),
            }
        }
        fn peer_desc(&self) -> String { self.peer.clone() }
    }

    /// Serve one TCP connection through the shared kernel (design 决策 #1/#4/#5):
    /// NDJSON frames → handle_request → response; registers into ConnectionRegistry on
    /// connect and unregisters (implicit unsubscribe + session_owner release) on disconnect.
    async fn serve_connection(
        mut conn: TcpConnection, server: &Arc<IpcServer>, token: Option<&str>,
    ) -> Result<(), IpcError> {
        let conn_id = server.registry.register(Box::new(conn));
        // --token auth: checked at initialize-time (design 错误处理); reject early.
        if let Some(expected) = token {
            if !initialize_authenticates(&conn_id, server, expected).await {
                server.registry.unregister(&conn_id);
                return Ok(()); // -32806 sent by initialize_authenticates
            }
        }
        loop {
            let line = match conn.recv().await {
                Some(Ok(line)) => line,
                Some(Err(e)) => { server.registry.unregister(&conn_id); return Err(e); }
                None => break, // client closed
            };
            let trimmed = line.trim().to_string();
            if trimmed.is_empty() { continue; }
            let response = server.handle_request(&trimmed).await
                .unwrap_or_else(|e| format!(r#"{{"jsonrpc":"2.0","id":null,"error":{{"code":-32603,"message":"{e}"}}}}"#));
            server.registry.send_to(&conn_id, &response)?;
            if server.shutdown_requested() { break; }
        }
        // disconnect: implicit unsubscribe + owner release (design 错误处理: subscribe 断开隐式退订)
        server.registry.unregister(&conn_id);
        Ok(())
    }

    /// Check the initialize handshake token; on mismatch reply -32806 and return false.
    async fn initialize_authenticates(conn_id: &str, server: &Arc<IpcServer>, expected: &str) -> bool {
        // v0.4 implementation detail: read the client's initialize request (or the first
        // request) and verify its `token` param matches; reply
        // {"jsonrpc":"2.0","id":..,"error":{"code":-32806,"message":"NotAuthorized"}} on failure.
        let _ = (conn_id, server, expected); // placeholder signature — full body in v0.4 impl
        true
    }

            let mut map = self.sessions.lock().unwrap();

            usage.completion_tokens += completion_tokens;

            self.sessions.lock().unwrap().get(session).copied()

            // manual fold — SessionUsage deliberately derives no Sum
            let mut total = SessionUsage::default();
            for usage in self.sessions.lock().unwrap().values() {
                total.prompt_tokens += usage.prompt_tokens;
                total.completion_tokens += usage.completion_tokens;
                total.turns += usage.turns;
            }
            total
        }
    }

        fn default() -> Self { Self::new() }
